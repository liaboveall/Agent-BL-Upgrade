import {chromium} from './browser/node_modules/playwright/index.mjs';
import fs from 'node:fs/promises';
const base='http://127.0.0.1:4178',dir='/tmp/agentbl-review-20261017';
const browser=await chromium.launch({executablePath:'/home/mars/.cache/ms-playwright/chromium-1243/chrome-linux64/chrome',args:['--no-sandbox']});
const context=await browser.newContext({viewport:{width:1440,height:1000},locale:'en-US'}),page=await context.newPage();
const out={errors:[],approvals:[]};page.on('pageerror',e=>out.errors.push(e.message));
try {
 await page.goto(base+'/#/agent',{waitUntil:'networkidle'});
 if((await page.locator('.mode-pill').innerText()).trim()!=='DEMO')throw new Error('Refuse non-Demo mode');
 const queue=await (await page.request.get(base+'/api/agent/approvals')).json();
 for(const a of queue.approvals.filter(a=>a.status==='PENDING')){
  if(a.mode!=='demo'||a.route!=='sim')throw new Error('Refuse non-sim approval');
  await page.locator(`[data-focus-key="approve:${a.approval_id}"]`).click();
  out.approvals.push({id:a.approval_id,action:a.action,mode:a.mode,route:a.route});
 }
 await page.waitForTimeout(1200);
 out.pool=await (await page.request.get(base+'/api/pool/status?pool_id=CASE-EBL-CU-SIN-SHA')).json();
 out.approvalResult=await (await page.request.get(base+'/api/agent/approvals')).json();
 await page.screenshot({path:dir+'/agent-settled-english.png',fullPage:true});
 await page.locator('#tab-discover').click();
 await page.locator('#mystery-preview-btn').click();
 await page.locator('#mystery-open-btn').waitFor({state:'visible'});
 out.mysteryPreview=await page.locator('#mystery-modal').innerText();
 await page.locator('#mystery-open-btn').click();
 await page.locator('#mystery-proof-status').waitFor({state:'visible',timeout:20000});
 await page.waitForTimeout(600);
 out.mysteryProof=await page.locator('#mystery-proof-status').innerText();
 out.mysteryResult=await page.locator('#mystery-modal').innerText();
 await page.screenshot({path:dir+'/mystery-proof-english.png',fullPage:true});
 await page.locator('#mystery-claim-discovery-btn').click();
 await page.waitForTimeout(600);
 out.passportClaim=await page.locator('#mystery-claim-discovery-btn').innerText();
 await page.keyboard.press('Escape');
 out.passportCount=await page.locator('#passport-collection-count').innerText();
} catch(e){out.harnessError=e.message;}
await fs.writeFile(dir+'/browser-finish.json',JSON.stringify(out,null,2));
console.log(JSON.stringify({errors:out.errors,harnessError:out.harnessError,approvals:out.approvals,pool:out.pool,mysteryProof:out.mysteryProof,passportCount:out.passportCount},null,2));
await browser.close();
