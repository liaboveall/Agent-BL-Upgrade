import {chromium} from './browser/node_modules/playwright/index.mjs';
import fs from 'node:fs/promises';
const dir='/tmp/agentbl-review-20261017',base='http://127.0.0.1:4177';
const browser=await chromium.launch({executablePath:'/home/mars/.cache/ms-playwright/chromium-1243/chrome-linux64/chrome',args:['--no-sandbox']});
const context=await browser.newContext({viewport:{width:1440,height:1000}});const page=await context.newPage();
const result={errors:[]};page.on('pageerror',e=>result.errors.push(e.message));
await page.goto(base,{waitUntil:'networkidle'});
await page.locator('#market-subscription').fill('100000');
const subscribed=page.waitForResponse(r=>r.url().endsWith('/api/pool/subscribe')&&r.request().method()==='POST');
const beforeText=await page.locator('#market-detail').innerText();
await page.locator('#market-subscribe-btn').click();
const response=await (await subscribed).json();
result.subscription={beforeText,response};
await page.waitForTimeout(300);
await page.screenshot({path:dir+'/subscription.png',fullPage:true});
const pool=await (await page.request.get(base+'/api/pool/status?pool_id=CASE-EBL-CU-SIN-SHA')).json();
await page.locator('#market-subscription').fill(String(pool.targetUsd-pool.subscribedUsd));
const funded=page.waitForResponse(r=>r.url().endsWith('/api/pool/subscribe')&&r.request().method()==='POST');
await page.locator('#market-subscribe-btn').click();await funded;await page.waitForTimeout(700);
result.funded={buttonEnabled:await page.locator('#market-subscribe-btn').isEnabled(),detail:await page.locator('#market-detail').innerText(),card:await page.locator('[data-case="CASE-EBL-CU-SIN-SHA"]').innerText()};
await page.screenshot({path:dir+'/funded-still-open.png',fullPage:true});
await page.locator('#tab-mint').click();
await page.route('**/api/agent/cases/CASE-EBL-GASOIL-SIN-JKT/preferences',async route=>{await new Promise(r=>setTimeout(r,1800));await route.continue();});
await page.locator('#case-dropdown').selectOption('CASE-EBL-GASOIL-SIN-JKT');
result.switchRace={selected:await page.locator('#case-dropdown').inputValue(),strip:await page.locator('.deal-strip').innerText(),mintEnabled:await page.locator('#mint-btn').isEnabled()};
await page.screenshot({path:dir+'/case-switch-stale.png',fullPage:true});
await page.waitForTimeout(2200);await page.unroute('**/api/agent/cases/CASE-EBL-GASOIL-SIN-JKT/preferences');
await page.locator('#case-dropdown').selectOption('CASE-EBL-CU-SIN-SHA');await page.waitForTimeout(250);
await page.locator('#pref-min-price').fill('0.999');
const saved=page.waitForResponse(r=>r.url().includes('/preferences')&&r.request().method()==='PUT');
await page.locator('#pref-min-price').press('Tab');await saved;await page.waitForTimeout(500);
await page.locator('#case-dropdown').selectOption('CASE-EBL-GASOIL-SIN-JKT');await page.waitForTimeout(500);
result.preferenceBleed={shown:await page.locator('#pref-min-price').inputValue(),before:await (await page.request.get(base+'/api/agent/cases/CASE-EBL-GASOIL-SIN-JKT/preferences')).json()};
const saveB=page.waitForResponse(r=>r.url().includes('/preferences')&&r.request().method()==='PUT');
await page.locator('#pref-speed').selectOption('FAST');await saveB;
result.preferenceBleed.after=await (await page.request.get(base+'/api/agent/cases/CASE-EBL-GASOIL-SIN-JKT/preferences')).json();
await page.screenshot({path:dir+'/preference-cross-case.png',fullPage:true});
await page.locator('.lang-btn').click();
result.englishPages=[];
for(const view of ['market','mint','agent','discover']){
 await page.locator('#tab-'+view).click();await page.waitForTimeout(150);
 result.englishPages.push({view,text:await page.locator('main').innerText(),geometry:await page.evaluate(()=>({w:innerWidth,scrollWidth:document.documentElement.scrollWidth}))});
 await page.screenshot({path:dir+'/'+view+'-en-1440.png',fullPage:true});
}
await page.setViewportSize({width:1280,height:800});await page.locator('#tab-market').click();
result.desktop1280=await page.evaluate(()=>({w:innerWidth,scrollWidth:document.documentElement.scrollWidth}));
await page.screenshot({path:dir+'/market-en-1280.png',fullPage:true});
await fs.writeFile(dir+'/browser-flows.json',JSON.stringify(result,null,2));
console.log(JSON.stringify({switchRace:result.switchRace,preferenceBleed:result.preferenceBleed,funded:result.funded,desktop1280:result.desktop1280,errors:result.errors},null,2));
await browser.close();
